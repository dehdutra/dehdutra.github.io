// ============================================================ // 41. CAMADA DE MUNICÍPIOS — CARREGAMENTO DINÂMICO POR ESTADO // ============================================================ // // Os arquivos estão em: // // municipios_ranking/ // // Exemplos: // // AC_01.geojson // AM_01.geojson // AM_02.geojson // // A rotina abaixo: // 1. identifica a UF clicada; // 2. consulta a pasta do GitHub; // 3. encontra todos os arquivos UF_XX.geojson; // 4. carrega todos os blocos; // 5. une as feições; // 6. cria uma única camada municipal; // 7. aplica a mesma classificação de cores dos estados. // // ============================================================ // ============================================================ // 41.1 — CONFIGURAÇÃO DO GITHUB // ============================================================ const MUNICIPIOS_GITHUB_API = "https://api.github.com/repos/dehdutra/dehdutra.github.io/contents/municipios_ranking"; const MUNICIPIOS_GITHUB_RAW = "https://raw.githubusercontent.com/dehdutra/dehdutra.github.io/main/municipios_ranking/"; // ============================================================ // 41.2 — GRUPO DOS MUNICÍPIOS // ============================================================ const grupoMunicipios = L.featureGroup(); // ============================================================ // 41.3 — CONTROLE DA CAMADA MUNICIPAL // ============================================================ let camadaMunicipiosAtual = null; let estadoMunicipiosAtual = null; let carregandoMunicipios = false; // ============================================================ // 41.4 — ADICIONAR MUNICÍPIOS AO CONTROLE DE CAMADAS // ============================================================ // // O controle já existe no seu código. // Aqui apenas acrescentamos a nova camada. // // ============================================================ controleCamadas.addOverlay( grupoMunicipios, "Municípios" ); // ============================================================ // 41.5 — OBTER ARQUIVOS DE UM ESTADO // ============================================================ async function obterArquivosMunicipiosEstado( sigla ) { const resposta = await fetch( MUNICIPIOS_GITHUB_API, { cache: "no-cache" } ); if ( !resposta.ok ) { throw new Error( `Não foi possível consultar a pasta municipios_ranking: HTTP ${resposta.status}` ); } const arquivos = await resposta.json(); const padrao = new RegExp( `^${sigla}_[0-9]+\\.geojson$`, "i" ); const encontrados = arquivos .filter( arquivo => arquivo.type === "file" && padrao.test( arquivo.name ) ) .sort( (a, b) => { const numeroA = parseInt( a.name.match( /_(\d+)\.geojson$/i )[1] ); const numeroB = parseInt( b.name.match( /_(\d+)\.geojson$/i )[1] ); return numeroA - numeroB; } ); return encontrados; } // ============================================================ // 41.6 — CARREGAR UM BLOCO MUNICIPAL // ============================================================ async function carregarBlocoMunicipios( arquivo ) { const url = MUNICIPIOS_GITHUB_RAW + arquivo.name; const resposta = await fetch( url, { cache: "force-cache" } ); if ( !resposta.ok ) { throw new Error( `${arquivo.name}: HTTP ${resposta.status}` ); } const bruto = await resposta.json(); const dados = normalizarGeoJSON( bruto ); if ( !dados.features || dados.features.length === 0 ) { console.warn( `${arquivo.name}: nenhuma feição encontrada` ); return []; } return dados.features; } // ============================================================ // 41.7 — ESTILO DOS MUNICÍPIOS // ============================================================ // // Usa exatamente a mesma classificação: // 1º #7f0000 // 2º #b30000 // 3º #d7301f // 4º #ef6548 // 5º #fdae61 // 6º–19º #bdbdbd // 20º #9ecae1 // 21º #6baed6 // 22º #4292c6 // 23º #2171b5 // 24º #084594 // // ============================================================ function estiloMunicipio( feature ) { const ranking = obterRanking( feature ); const temRanking = ranking !== null; return { color: "#555555", weight: 0.7, opacity: 0.95, fillColor: temRanking ? ( coresRanking[ranking] || "#bdbdbd" ) : "#9e9e9e", fillOpacity: temRanking ? 0.72 : 0.18 }; } // ============================================================ // 41.8 — IDENTIFICAR NOME DO MUNICÍPIO // ============================================================ // // Como os GeoJSON podem ter nomes diferentes, // usamos várias possibilidades. // // ============================================================ function obterNomeMunicipio( feature ) { return obterPropriedade( feature, [ "Municipality", "Municipio", "MUNICIPIO", "municipio", "NM_MUN", "NM_MUNICIP", "NAME", "Name", "name", "NOME", "Nome", "nome", "region_nam", "Region Name" ] ); } // ============================================================ // 41.9 — CRIAR CAMADA MUNICIPAL // ============================================================ function criarCamadaMunicipios( features, sigla ) { // -------------------------------------------------------- // Remove a camada municipal anterior // -------------------------------------------------------- if ( camadaMunicipiosAtual ) { grupoMunicipios.removeLayer( camadaMunicipiosAtual ); } // -------------------------------------------------------- // Cria a nova camada // -------------------------------------------------------- const dadosMunicipios = { type: "FeatureCollection", features: features }; const camada = L.geoJSON( dadosMunicipios, { style: function(feature) { return estiloMunicipio( feature ); }, onEachFeature: function( feature, layer ) { layer.on({ mouseover: function(e) { e.target.setStyle({ weight: 1.8, color: "#111827", fillOpacity: 0.88 }); }, mouseout: function(e) { camada.resetStyle( e.target ); }, click: function(e) { mostrarPopupMunicipio( e.target.feature, e.target, sigla ); atualizarPainelVetorMunicipio( e.target.feature, sigla ); } }); } } ); camadaMunicipiosAtual = camada; grupoMunicipios.addLayer( camada ); return camada; } // ============================================================ // 41.10 — POPUP DOS MUNICÍPIOS // ============================================================ function mostrarPopupMunicipio( feature, layer, sigla ) { const nome = obterNomeMunicipio( feature ) || "Município"; const ranking = obterRanking( feature ); const area = obterArea( feature ); let html = `
${escaparHTML(nome)}
Município — ${escaparHTML(sigla)}
`; if ( ranking !== null ) { html += `
Ranking: ${formatarNumero(ranking)}
`; } if ( area !== null ) { html += `
Área queimada: ${formatarNumero(area)} km²
`; } html += `
`; layer.bindPopup( html ).openPopup(); } // ============================================================ // 41.11 — CARREGAR MUNICÍPIOS DO ESTADO // ============================================================ async function carregarMunicipiosEstado( sigla ) { // -------------------------------------------------------- // Evita duas requisições simultâneas para a mesma UF // -------------------------------------------------------- if ( carregandoMunicipios && estadoMunicipiosAtual === sigla ) { return; } carregandoMunicipios = true; estadoMunicipiosAtual = sigla; try { console.log( `Carregando municípios de ${sigla}...` ); // ---------------------------------------------------- // 1. Descobrir os blocos // ---------------------------------------------------- const arquivos = await obterArquivosMunicipiosEstado( sigla ); if ( arquivos.length === 0 ) { console.warn( `Nenhum arquivo municipal encontrado para ${sigla}.` ); grupoMunicipios.clearLayers(); camadaMunicipiosAtual = null; return; } console.log( `Arquivos encontrados para ${sigla}:`, arquivos.map( arquivo => arquivo.name ) ); // ---------------------------------------------------- // 2. Carregar todos os blocos // ---------------------------------------------------- const resultados = await Promise.all( arquivos.map( arquivo => carregarBlocoMunicipios( arquivo ) ) ); // ---------------------------------------------------- // 3. UNIR todos os blocos // ---------------------------------------------------- const features = resultados.flat(); console.log( `${sigla}: ${features.length} municípios carregados.` ); // ---------------------------------------------------- // 4. Criar uma única camada // ---------------------------------------------------- criarCamadaMunicipios( features, sigla ); // ---------------------------------------------------- // 5. Mostrar automaticamente a camada // ---------------------------------------------------- if ( !map.hasLayer( grupoMunicipios ) ) { grupoMunicipios.addTo( map ); } // ---------------------------------------------------- // 6. Atualizar controle // ---------------------------------------------------- console.log( `Camada municipal de ${sigla} adicionada ao mapa.` ); } catch (erro) { console.error( `Erro ao carregar municípios de ${sigla}:`, erro ); grupoMunicipios.clearLayers(); camadaMunicipiosAtual = null; } finally { carregandoMunicipios = false; } } // ============================================================ // 41.12 — PAINEL DOS MUNICÍPIOS // ============================================================ // // Reutiliza o mesmo painel já existente. // Portanto não cria outro painel na interface. // // ============================================================ function atualizarPainelVetorMunicipio( feature, sigla ) { const nomeElemento = document.getElementById( "vetor-nome" ); const tipoElemento = document.getElementById( "vetor-tipo" ); const informacoes = document.getElementById( "vetor-informacoes" ); const vazio = document.getElementById( "painel-vetor-vazio" ); const nome = obterNomeMunicipio( feature ) || "Município"; nomeElemento.innerText = nome; tipoElemento.innerText = `Município — ${sigla}`; informacoes.innerHTML = ""; let quantidadeInformada = 0; CAMPOS_VETOR.forEach( campo => { const valor = obterValorCampoVetor( feature, campo.chave ); if ( valor !== null && valor !== undefined && valor !== "" ) { quantidadeInformada++; } const item = document.createElement( "div" ); item.className = "vetor-info-item"; const label = document.createElement( "div" ); label.className = "vetor-info-label"; label.innerText = campo.rotulo; const valorElemento = document.createElement( "div" ); valorElemento.className = "vetor-info-valor"; if ( campo.classe ) { valorElemento.classList.add( campo.classe ); } valorElemento.innerText = formatarValorCampo( valor, campo ); item.appendChild( label ); item.appendChild( valorElemento ); informacoes.appendChild( item ); } ); if ( quantidadeInformada > 0 ) { vazio.style.display = "none"; } else { vazio.style.display = "block"; vazio.innerText = "Os campos de ranqueamento não foram encontrados nesta feição."; } const painel = document.getElementById( "painel-vetor" ); painel.classList.remove( "recolhido" ); document .getElementById( "painel-vetor-seta" ) .innerText = "▼"; } // ============================================================ // 41.13 — DETECTAR CLIQUE NO ESTADO // ============================================================ // // IMPORTANTE: // // Não alteramos o seu evento click existente. // // O Leaflet propaga o evento do polígono para o // FeatureGroup. Portanto usamos este listener adicional. // // Assim, quando você clicar no estado: // // Acre -> carrega AC_01 // // Amazonas -> carrega AM_01 + AM_02 + ... // // São Paulo -> carrega SP_01 + SP_02 + ... // // ============================================================ grupoEstados.on( "click", function(event) { const feature = event.layer?.feature; if ( !feature ) { return; } const sigla = obterSiglaEstado( feature ); if ( !sigla ) { console.warn( "Não foi possível identificar a UF clicada." ); return; } console.log( `Estado clicado: ${sigla}` ); carregarMunicipiosEstado( sigla ); } ); // ============================================================ // 41.14 — CLIQUE NO CONTROLE "ESTADOS" // ============================================================ // // Quando o usuário desmarca Estados, também escondemos // os municípios. // // ============================================================ map.on( "overlayremove", function(event) { if ( event.layer === grupoEstados ) { if ( map.hasLayer( grupoMunicipios ) ) { map.removeLayer( grupoMunicipios ); } } } ); // ============================================================ // 41.15 — QUANDO ESTADOS VOLTAR PARA O MAPA // ============================================================ // // Se já existe uma camada municipal carregada, // ela pode voltar automaticamente. // // ============================================================ map.on( "overlayadd", function(event) { if ( event.layer === grupoEstados ) { if ( camadaMunicipiosAtual && estadoMunicipiosAtual ) { grupoMunicipios.addTo( map ); } } } );
